# Both platforms MUST produce byte-identical outputs (expected_role_byte,
# expected_session_id, expected_sig) given these exact inputs. Ed25519 is

# layout. The Android pinning lives in

bob_signing_seed      = 000102030405060708090a0b0c0d0e0f101112131415161718191a1b1c1d1e1f
bob_signing_pub       = 03a107bff3ce10be1d70dd18e74bc09967e4d6309ba50d5f1ddc8664125531b8

# bob_pq_pub: 1184 bytes (ML-KEM-768 public-key length). For test purposes
# we use a deterministic synthetic value so the vector is reproducible
# without depending on a specific KEM key generation:
#   for i in 0 .. 1183: bob_pq_pub[i] = (i XOR 0xA5) & 0xFF
bob_pq_pub_formula    = bob_pq_pub[i] = (i ^ 0xA5) & 0xFF, for i in 0..1183

# Bob's eph is lex-greater than Alice's (compares unsigned), so Bob is
# the "higher" side and gets role byte 0x02.
expected_role_byte    = 02

# BLAKE2b-256 keyed with "ZERION_HANDSHAKE_SESSION_v1" (27 ASCII bytes,
# no NUL) over sortedConcat(min(alice_eph, bob_eph), max(...)).
expected_session_id   = 9ddaa2c9b20ee986425a94bd5c8301a59e8910358890da831e620e93fa93cf0a
