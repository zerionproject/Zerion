# ============================================================================
# LOGGING SWEEP — strip all logging from release APK
# ============================================================================
# java.util.logging.Logger calls in vendored upstream code (onionwrapper-*)
# and android.util.Log calls anywhere become no-ops at compile time. R8 then
# removes the call sites and any string literals constructed only for the
# arguments. Combined with the runtime LogManager reset in
# ZerionApplicationImpl, this provides defence in depth — strings cannot be
# recovered from a decompiled release APK.
-assumenosideeffects class java.util.logging.Logger {
    public void info(...);
    public void warning(...);
    public void severe(...);
    public void fine(...);
    public void finer(...);
    public void finest(...);
    public void config(...);
    public void log(...);
    public void logp(...);
    public void entering(...);
    public void exiting(...);
    public void throwing(...);
}
-assumenosideeffects class android.util.Log {
    public static *** d(...);
    public static *** v(...);
    public static *** i(...);
    public static *** w(...);
    public static *** e(...);
    public static *** wtf(...);
}
# The embedded I2P router ships its own logger (net.i2p.util.Log). It is muted
# at runtime by logger.defaultLevel=CRIT + logRotationLimit=0 in BundledI2pRouter,
# so no router/transport metadata is written on device. Do NOT add
# -assumenosideeffects for net.i2p.util.Log: R8 then deletes those call sites and
# the router's logging is not cleanly side-effect-free, which breaks router
# startup (I2P never reaches running) in the minified release build.
